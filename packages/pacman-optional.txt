# Optional packages. Nothing in the config depends on these — each one enables
# a feature that is commented out or degraded without it.
#
#   ./install.sh --optional
#
# Install them one at a time as you need them rather than all at once: every
# GUI package here adds an entry to the wofi application list.

playerctl          # media keybinds in conf/keybinds.lua (commented out)
brightnessctl      # backlight keybinds in conf/keybinds.lua (commented out); laptops only
pavucontrol        # GUI volume mixer; re-add "on-click" to waybar's pulseaudio module
nvtop              # GPU-only monitor, more detail than btop's GPU panel
qt5-wayland        # only if you run Qt5 apps
pipewire-alsa      # only if you run apps that talk to ALSA directly
noto-fonts         # broad Unicode coverage outside the Nerd Font
noto-fonts-emoji   # without it, emoji render as empty boxes
