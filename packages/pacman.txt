# Core packages — the hard dependencies of this config.
# Nothing here is optional: remove any of it and something in the config stops
# working. Extras live in pacman-optional.txt.
#
#   ./install.sh --packages
#
# Lines starting with # and blank lines are ignored.

# Compositor
hyprland
xorg-xwayland                 # X11 apps; without it they simply don't start

# Session plumbing
xdg-desktop-portal-hyprland   # screen sharing, native file pickers
xdg-desktop-portal-gtk        # GTK portal backend (file dialogs)
polkit
hyprpolkitagent               # started from conf/autostart.lua

# The stack itself — every one of these is referenced by the config
waybar
wofi
kitty
mako
hyprpaper                     # wallpaper, started from conf/autostart.lua
hyprlock                      # screen lock, SUPER+L
hypridle                      # locks and blanks on idle
btop                          # resource monitor on SUPER+Escape

# Screenshots (Print / SHIFT+Print / ALT+Print) and clipboard
grim
slurp
wl-clipboard
cliphist                      # clipboard history on SUPER+SHIFT+V
libnotify                     # notify-send, used by the screenshot script
jq                            # the screenshot script reads window geometry with it

# Dark GTK apps
adwaita-icon-theme            # icons and the Adwaita cursor
dconf                         # gsettings needs a backend to persist anything

# Audio. wpctl comes from wireplumber and drives the volume keybinds.
pipewire
pipewire-pulse
wireplumber

# Wayland support for Qt6 apps
qt6-wayland

# Referenced by name in waybar, kitty, wofi and mako configs
ttf-jetbrains-mono-nerd
